  local name="$1" oracle="$2"
  shift 2
  target/debug/wallet-conformance vectors --adapter "$oracle" --target http://127.0.0.1:18790/v2/ --environment source-2fc5c383-threshold-internals --timeout-seconds 60 --report "/tmp/wc-threshold-internals-ypH6gd/$name.json" "$@"
}

 --operation sha256 --operation sha512 --operation blake2b --operation ed25519.sign --operation ed25519.verify --operation passphrase.hash-pbkdf2 --operation passphrase.check-pbkdf2 --operation passphrase.check-scrypt --operation keys.verify-extended --operation coin-selection.select --operation coin-selection.haskell --operation root-key.parse --operation root-key.create-v2 --operation root-key.check-v2 --operation root-key.sign-v2 --operation root-key.change-v2 --operation root-key.derive-v2
